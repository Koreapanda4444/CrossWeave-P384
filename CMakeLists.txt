cmake_minimum_required(VERSION 3.20)
project(CrossWeaveP384 VERSION 0.1 LANGUAGES C)

option(CRYPTO_BUILD_RESEARCH "Build bounded research tools when implemented" OFF)

option(CRYPTO_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(cw_options INTERFACE)
if(MSVC)
  target_compile_options(cw_options INTERFACE /W4 /WX)
else()
  target_compile_options(cw_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(CRYPTO_ENABLE_SANITIZERS)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
    target_compile_options(cw_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cw_options INTERFACE -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "Sanitizers require a supported GCC/Clang environment")
  endif()
endif()
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  function(cw_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE cw_options OpenSSL::Crypto)
    target_include_directories(${name} PRIVATE src include tests)
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  cw_test(environment tests/environment.c)
endif()
